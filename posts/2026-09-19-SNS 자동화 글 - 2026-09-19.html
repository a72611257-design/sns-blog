from datetime import datetime
from html import escape
import os

text = open("post.txt", "r", encoding="utf-8").read()
lines = text.splitlines()

title = next((x.replace("제목:", "", 1).strip() for x in lines if x.startswith("제목:")), "")
content = "\n".join(x for x in lines if not x.startswith("제목:") and not x.startswith("본문:")).strip()

today = datetime.now().strftime("%Y-%m-%d")
title = title or ("SNS 자동화 글 - " + today)

os.makedirs("posts", exist_ok=True)

safe = "".join(x for x in title if x.isalnum() or x in " -_").strip() or "blog-post"
name = today + "-" + safe
filename = os.path.join("posts", name + ".html")

n = 2
while os.path.exists(filename):
filename = os.path.join("posts", name + "-" + str(n) + ".html")
n += 1

page = """<!DOCTYPE html>
<html lang="ko">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>""" + escape(title) + """</title>
<style>
body{max-width:800px;margin:40px auto;padding:20px;font-family:Arial,sans-serif;line-height:1.8}
h1{border-bottom:2px solid #333;padding-bottom:10px}
.date{color:#777}
</style>
</head>
<body>
<h1>""" + escape(title) + """</h1>
<div class="date">작성일: """ + today + """</div>
<div>""" + escape(content).replace("\n", "<br>") + """</div>
<hr>
<p>이 글은 Python을 이용한 SNS 자동화 테스트입니다.</p>
</body>
</html>"""

open(filename, "w", encoding="utf-8").write(page)

files = [x for x in os.listdir("posts") if x.endswith(".html")]
files.sort(reverse=True)

links = "".join(
'<div><h2><a href="posts/' + escape(x) + '">' + escape(x[:-5]) + '</a></h2></div>'
for x in files
)

index = """<!DOCTYPE html>
<html lang="ko">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>나의 자동화 블로그</title>
<style>
body{max-width:800px;margin:40px auto;padding:20px;font-family:Arial,sans-serif;line-height:1.8}
h1{border-bottom:2px solid #333;padding-bottom:10px}
a{text-decoration:none}
</style>
</head>
<body>
<h1>나의 자동화 블로그</h1>
<p>자동으로 발행되는 블로그입니다.</p>
""" + links + """
</body>
</html>"""

open("index.html", "w", encoding="utf-8").write(index)

print("========================================")
print("다중 글 블로그 생성 완료!")
print("제목:", title)
print("저장 파일:", filename)
print("현재 글 개수:", len(files))
print("========================================")
